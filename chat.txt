namespace Vendor\Settings\Resolvers;


class ValueCaster
{
    public static function cast(string $type, mixed $value): mixed
    {
        return match ($type) {
            'integer' => (int) $value,
            'float' => (float) $value,
            'boolean' => filter_var($value, FILTER_VALIDATE_BOOL),
            'array' => (array) $value,
            'object' => (object) $value,
            'json' => $value,
            'string' => (string) $value,
            default => $value,
        };
    }
}
Resolver: ScopeHierarchyResolver.php
<?php


namespace Vendor\Settings\Resolvers;


use Vendor\Settings\Contracts\SettingsScope;


class ScopeHierarchyResolver
{
    public function resolve(?SettingsScope $scope): array
    {
        if (! $scope) {
            return [
                [
                    'scope_type' => 'global',
                    'scope_id' => null,
                ],
            ];
        }


        $resolved = [
            [
                'scope_type' => $scope->getSettingsScopeType(),
                'scope_id' => (string) $scope->getSettingsScopeId(),
            ],
        ];


        foreach ($scope->getSettingsFallbackScopes() as $fallback) {
            $resolved[] = [
                'scope_type' => $fallback->getSettingsScopeType(),
                'scope_id' => (string) $fallback->getSettingsScopeId(),
            ];
        }


        $resolved[] = [
            'scope_type' => 'global',
            'scope_id' => null,
        ];


        return $resolved;
    }
}

Repository: CachedSettingsRepository.php
<?php


namespace Vendor\Settings\Repositories;


use Illuminate\Support\Facades\Cache;
use Vendor\Settings\Contracts\SettingsRepository;


class CachedSettingsRepository implements SettingsRepository
{
    public function __construct(
        protected SettingsRepository $repository,
        protected string $cachePrefix = 'settings',
        protected int $ttl = 3600,
    ) {}


    public function get(string $key, mixed $default = null): mixed
    {
        return Cache::remember(
            $this->cacheKey($key),
            $this->ttl,
            fn () => $this->repository->get($key, $default)
        );
    }


    public function set(string $key, mixed $value): void
    {
        $this->repository->set($key, $value);


        Cache::forget($this->cacheKey($key));
    }


    public function has(string $key): bool
    {
        return $this->repository->has($key);
    }


    public function forget(string $key): void
    {
        $this->repository->forget($key);


        Cache::forget($this->cacheKey($key));
    }


    public function all(?string $prefix = null): array
    {
        return $this->repository->all($prefix);
    }


    protected function cacheKey(string $key): string
    {
        return $this->cachePrefix . ':' . $key;
    }
}
Manager: SettingsManager.php
<?php
use Vendor\Settings\Repositories\DatabaseSettingsRepository;


class SettingsManager implements SettingsManagerContract
{
    protected ?SettingsScope $scope = null;


    protected SettingsRepository $repository;


    public function __construct()
    {
        $this->repository = new CachedSettingsRepository(
            new DatabaseSettingsRepository()
        );
    }


    public function scope(?SettingsScope $scope): static
    {
        $clone = clone $this;


        $clone->scope = $scope;


        $clone->repository = new CachedSettingsRepository(
            new DatabaseSettingsRepository($scope)
        );


        return $clone;
    }


    public function get(string $key, mixed $default = null): mixed
    {
        if ($this->hasSubtree($key)) {
            return $this->all($key);
        }


        return $this->repository->get($key, $default);
    }


    public function set(string $key, mixed $value): static
    {
        $this->repository->set($key, $value);


        return $this;
    }


    public function all(?string $prefix = null): array
    {
        return $this->repository->all($prefix);
    }


    public function has(string $key): bool
    {
        return $this->repository->has($key);
    }


    public function forget(string $key): void
    {
        $this->repository->forget($key);
    }


    protected function hasSubtree(string $key): bool
    {
        return \Vendor\Settings\Models\SettingDefinition::query()
            ->where('key', 'LIKE', $key . '.%')
            ->exists();
    }
}
